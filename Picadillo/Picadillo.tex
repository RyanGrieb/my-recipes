\makeRecipeTitle{Picadillo}

\section*{Ingredients}

\begin{itemize}
      \item Olive Oil – 1 tablespoon
      \item Yellow Potatoes – 8 ounces (about 3--4 small), diced
      \item Yellow Onion – 1 medium, chopped
      \item Jalapeño Pepper – 1, chopped
      \item Serrano Pepper – 1, chopped (optional, for spicier)
      \item Garlic – 2 cloves, chopped
      \item Ground Beef – 1 pound (other ground meats like chicken, turkey, or pork work too)
      \item Large Tomatoes – 2, chopped
      \item Beef Stock – 3/4 cup (or chicken/vegetable stock)
      \item Paprika – 1 tablespoon
      \item Mexican Oregano – 1 tablespoon
      \item Cumin – 1 teaspoon
      \item Salt and Pepper – to taste
      \item For Serving – Cooked rice or warmed tortillas, sliced peppers, chili flakes, fresh lime juice, hot sauce
\end{itemize}

\newpage
\makeRecipeTitle{Picadillo}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Sauté the Vegetables} \\
            Heat the oil in a large pan over medium heat. Add the potatoes, onion, and peppers. Cook for 5 minutes, stirring occasionally, until they begin to soften.

      \item \textbf{Brown the Beef} \\
            Stir in the garlic and ground beef. Break the beef apart with a wooden spoon and cook for about 5 minutes until browned; it does not need to be cooked all the way through at this stage.

      \item \textbf{Add the Tomatoes and Stock} \\
            Stir in the chopped tomatoes, beef stock, paprika, Mexican oregano, cumin, salt, and pepper. Reduce the heat to a simmer.

      \item \textbf{Simmer} \\
            Simmer gently for 20 minutes, stirring from time to time, until the potatoes are tender and the flavors have melded.

      \item \textbf{Serve} \\
            Spoon the picadillo over cooked rice or serve with warm tortillas. Finish with extra sliced peppers, chili flakes, a squeeze of fresh lime juice, and your favorite hot sauce.

\end{enumerate}
