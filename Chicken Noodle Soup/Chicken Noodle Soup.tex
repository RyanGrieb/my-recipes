\makeRecipeTitle{Chicken Noodle Soup}
\recipeServings{6}

\section*{Ingredients}

\begin{itemize}
      \item Boneless Chicken Breasts – 2 (about 1.4 lbs)
      \item \recipeLink{Frozen Chicken Stock}{Frozen Chicken Stock} – 8 cups (two 4-cup portions), thawed
      \item[\cut] Yellow Onion – 1, diced
      \item[\cut] Carrots – 3, sliced into coins
      \item[\cut] Celery Ribs – 3, sliced
      \item[\cut] Garlic Cloves – 3, minced
      \item[\cut] Fresh Parsley – ¼ cup, chopped
      \item[\measure] Dried Thyme – 1 tsp
      \item[\measure] Kosher Salt – 2 tsp*
      \item[\measure] Lemon Juice – 1 tbsp
      \item Bay Leaf – 1
      \item Egg Noodles – 6 oz (about 3 cups dry)
      \item Unsalted Butter – 2 tbsp
      \item Black Pepper – to taste
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item[*] \textbf{Salt:} Homemade stock is unsalted, so this soup needs more salt than you'd expect. Start with 2 tsp, then taste at the end and add more until it tastes like soup instead of hot water.
      \item \textbf{Meal Prep:} If the soup will sit for days, cook the noodles separately and add them to each bowl. Noodles stored in the broth soak it up and go soft.
\end{itemize}

\makeRecipeTitle{Chicken Noodle Soup}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything} \\
            Dice the onion, slice the carrots and celery, mince the garlic, and chop the parsley. Squeeze the lemon, and measure out everything marked with a spoon.

      \item \textbf{Sauté the Vegetables} \\
            Melt the butter in a large pot over medium heat. Add the onion, carrots, and celery and cook, stirring occasionally, until the onion is soft (6--8 minutes). Stir in the garlic and thyme until fragrant (30 seconds).

      \item \textbf{Poach the Chicken} \\
            Add the stock, bay leaf, and salt, and bring to a simmer. Add the whole chicken breasts, lower the heat, and simmer gently until they reach 165$^\circ$F in the thickest part (20--25 minutes). Move them to a cutting board.

      \item \textbf{Cook the Noodles} \\
            Bring the soup back to a boil, add the noodles, and cook until tender (6--8 minutes). While they cook, shred the chicken with two forks.

      \item \textbf{Serve} \\
            Take out the bay leaf. Stir the chicken back in along with the lemon juice and parsley. Taste, then season with salt and pepper until it's bright and well seasoned. Enjoy!

\end{enumerate}
