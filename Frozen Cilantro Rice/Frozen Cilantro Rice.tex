\makeRecipeTitle{Frozen Cilantro Rice}

\section*{Ingredients}

\begin{itemize}
      \item Jasmine or Basmati Rice – 2 cups
      \item Water – 2 cups
      \item Chicken Broth – 1 cup (Kettle\&Fire or similar)
      \item Fresh Cilantro – 1 cup, chopped
      \item Lime – 1 (juiced)
\end{itemize}

\newpage
\makeRecipeTitle{Frozen Cilantro Rice}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Rinse the Rice} \\
            Measure the rice, then rinse it under cold water until the water runs clear.

      \item \textbf{Cook the Rice} \\
            Add the rinsed rice, water, and chicken broth to a rice cooker and run the regular white rice setting.

      \item \textbf{Cool Slightly} \\
            Remove the cooked rice from the cooker and fluff it to release steam. Let it cool until it is warm rather than piping hot, so the cilantro does not cook and turn dull.

      \item \textbf{Add Cilantro and Lime} \\
            Stir in the chopped cilantro and the lime juice until evenly incorporated.

      \item \textbf{Portion and Freeze} \\
            Ladle into silicone molds and freeze overnight.

\end{enumerate}
