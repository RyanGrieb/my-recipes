\makeRecipeTitle{Loaded Potato Soup}

\section*{Ingredients}

\begin{itemize}
      \item Bacon, uncooked – 6 strips, cut into small pieces
      \item Butter, unsalted or salted – 3 Tablespoons
      \item Yellow Onion, chopped – 1 medium (about 1 1/2 cups/200g)
      \item Garlic – 3 large cloves, minced
      \item All-Purpose Flour – 1/3 cup (42g)
      \item Gold Potatoes, peeled and diced – 2 1/2 lbs (pieces no larger than 1'', about 6 large/1.15kg)
      \item Chicken Broth – 4 cups (945ml)
      \item Milk – 2 cups (475ml)
      \item Heavy Cream – 2/3 cup (155ml)
      \item Salt – 1 1/2 teaspoon* (see Notes)
      \item Ground Black Pepper – 1 teaspoon
      \item Ancho Chili Powder – 1/4--1/2 teaspoon** (see Notes)
      \item Sour Cream – 2/3 cup (160g)
      \item For Topping (optional) – Shredded cheddar cheese, chopped chives, additional sour cream, and crumbled bacon
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item[*] \textbf{Salt:} Start with 1 1/2 teaspoons, then taste and add more if needed. The amount will vary based on the brand of broth you use.
      \item[**] \textbf{Ancho Chili Powder:} If you haven't used it before, start with 1/4 teaspoon and taste the soup before adding up to 1/2 teaspoon, if you enjoy the flavor.
\end{itemize}

\makeRecipeTitle{Loaded Potato Soup}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Cook the Bacon} \\
            Place the bacon pieces in a large Dutch oven or soup pot over medium heat. Cook until crisp and browned. Remove the bacon pieces with a slotted spoon and set aside, leaving the rendered fat in the pot.

      \item \textbf{Sauté Aromatics} \\
            Add the butter and chopped onion to the pot. Cook over medium heat, stirring occasionally, until the onions are tender (3--5 minutes). Add the minced garlic and cook until fragrant, about 30 seconds.

      \item \textbf{Make the Roux} \\
            Sprinkle the flour over the onion mixture. Stir with a whisk or wooden spoon until smooth and no dry flour remains.

      \item \textbf{Add Potatoes and Liquids} \\
            Add the diced potatoes, chicken broth, milk, heavy cream, salt, pepper, and ancho chili powder. Stir everything together well.

      \item \textbf{Boil and Cook Potatoes} \\
            Bring the soup to a boil, then cook until the potatoes are tender when pierced with a fork, about 10 minutes.

      \item \textbf{Blend Half the Soup} \\
            Reduce the heat to a simmer. Carefully remove about half of the soup (roughly 5 cups) to a blender. Purée until smooth, then return the puréed soup to the pot. (Alternatively, use an immersion blender directly in the pot to partially blend the soup, leaving some chunks.)

      \item \textbf{Finish the Soup} \\
            Stir in the sour cream and the reserved bacon pieces. Let the soup simmer gently for 15 minutes before serving.

      \item \textbf{Serve} \\
            Ladle into bowls and top with extra sour cream, bacon, cheddar cheese, and chives as desired. Enjoy!

\end{enumerate}
