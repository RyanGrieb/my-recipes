\makeRecipeTitle{Oven Lemon Pepper Wings \& Drums}
\recipeServings{3--4}

\section*{Ingredients}

\textbf{The Chicken}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Chicken Wings – 4 (from 2 birds), split into drumettes and flats
      \item Chicken Drumsticks – 4 (about 2.3 lbs total with the wings)
\end{itemize}

\textbf{The Coating}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\measure] Baking Powder – 2 tsp (aluminum-free, not baking soda)
      \item[\measure] Lemon Pepper Seasoning – 1 tbsp
      \item[\measure] Garlic Powder – 1 tsp
      \item[\measure] Kosher Salt – ½ tsp
      \item[\measure] MSG – ½ tsp (optional but recommended)
\end{itemize}

\textbf{The Finish}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Unsalted Butter – 3 tbsp
      \item[\measure] Lemon Pepper Seasoning – 2 tbsp
      \item[\veg] Lemon – 1, zested, then cut into wedges for serving
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item \textbf{Baking Powder:} This is what crisps the skin in the oven. It raises the skin's pH so it browns and bubbles like fried skin. Measure it exactly; too much tastes metallic.
      \item \textbf{Wing Tips:} Save them in the stock bag.
\end{itemize}

\makeRecipeTitle{Oven Lemon Pepper Wings \& Drums}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything} \\
            Split the wings at the joints into drumettes and flats. Pat all the chicken extremely dry with paper towels. Zest the lemon and cut it into wedges, and measure out everything marked with a spoon.

      \item \textbf{The Coat} \\
            In a large bowl, mix the baking powder, 1 tbsp lemon pepper, garlic powder, salt, and MSG. Add the chicken and toss until every piece has a thin, even coating with no clumps.

      \item \textbf{Set Up the Rack} \\
            Line a sheet pan with foil and set a wire rack on top. Arrange the chicken skin-side up with space between the pieces. \textbf{Chef's Tip:} If you have time, refrigerate the tray uncovered overnight. The drier skin crisps even more.

      \item \textbf{The Low Bake} \\
            Put the pan on the upper-middle rack and bake at 250$^\circ$F (30 minutes). This slowly renders the fat out from under the skin.

      \item \textbf{The High Bake} \\
            Without taking the pan out, raise the oven to 425$^\circ$F and bake until the skin is deep golden and crackly (45--55 minutes). The wings usually finish first, so pull them as they're ready. The drums are done at 175$^\circ$F or higher at the bone.

      \item \textbf{The Finish} \\
            Melt the butter in a large bowl and stir in the 2 tbsp lemon pepper and the lemon zest. Add the hot chicken straight from the oven and toss to coat. The heat blooms the pepper. Serve with the lemon wedges. Enjoy!

\end{enumerate}
