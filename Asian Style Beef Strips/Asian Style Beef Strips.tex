\makeRecipeTitle{Asian Style Beef Strips}
\recipeServings{3--4}

\section*{Ingredients}

% UPDATED: Uses \raisebox to shift the bullet down (-0.25ex) for perfect vertical centering
\begin{itemize}
      \item Steak (flank preferably) – 1 lb
      \item[\measure] Baking soda – 1 tsp
      \item[\measure] Brown sugar – 1/3 cup
      \item[\measure] Corn starch – 1/3 cup
      \item[\measure] Corn starch slurry (corn starch and water) – 1 tbsp
      \item[\veg] Garlic – 1 tsp
      \item[\veg] Ginger – 1 tsp
      \item[\measure] Salt – 1 tsp
      \item[\measure] Soy sauce – 1/4 cup
      \item[\veg] Thai chili and green beans (optional)
\end{itemize}

\newpage
\makeRecipeTitle{Asian Style Beef Strips}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything} \\
            Thinly cut your beef using proper knife safety, and mince the garlic and ginger. Mix it with salt, cornstarch, and baking soda, and set aside. Slice the Thai chili and trim the green beans (if using), and measure out everything marked with a spoon.

      \item \textbf{Prepare Sauce} \\
            Combine brown sugar and light soy sauce in a bowl or container.

      \item \textbf{Cook Beef} \\
            Fry beef in a pan until golden brown and cooked through. Set aside.

      \item \textbf{Make Sauce} \\
            In the same pan, fry the garlic and ginger until fragrant (30 seconds). Add the sauce mixture and the corn starch slurry.

      \item \textbf{Thicken} \\
            Stir the sauce until it thickens to your desired consistency.

      \item \textbf{Combine} \\
            Add the cooked beef, chili, and green beans (if using) back to the sauce and mix until well coated.

\end{enumerate}
