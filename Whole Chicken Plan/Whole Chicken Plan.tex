\makeRecipeTitle{Whole Chicken Plan}
\recipeYield{a week of meals from 2 whole chickens}

\section*{Shopping}

\begin{itemize}
      \item H-E-B Natural Fresh Whole Chickens – 2 (about 5.3 lbs each, giblets and neck included)
\end{itemize}

\section*{Where Every Part Goes}

\textbf{Cook Tonight}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item \textbf{4 Wings + 4 Drumsticks} – \recipeLink{Lemon Pepper Wings and Drums}{Lemon Pepper Wings \& Drums}
\end{itemize}

\textbf{Freezer Bags}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item \textbf{4 Thighs}, boned and skinned (about 1.5 lbs) – \mainRecipeLink{Chipotle Meal Prep}
      \item \textbf{2 Breasts}, each sliced into 2 cutlets – \mainRecipeLink{Creamy Tuscan Chicken}
      \item \textbf{2 Breasts} (about 1.4 lbs) – \recipeLink{Chicken Noodle Soup}{Chicken Noodle Soup} (or the \mainRecipeLink{Chicken Vegetable Stir-Fry})
\end{itemize}

\textbf{The Stockpot}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item \textbf{2 Carcasses}, plus necks, hearts, gizzards, thigh bones, and wing tips – \recipeLink{Frozen Chicken Stock}{Frozen Chicken Stock} (about 12 cups)
\end{itemize}

\section*{Where the Stock Goes}

\begin{itemize}
      \item \textbf{8 cups} – \recipeLink{Chicken Noodle Soup}{Chicken Noodle Soup}
      \item \textbf{1 cup} – \mainRecipeLink{Chipotle Meal Prep} (the beans)
      \item \textbf{3 cups spare} – \mainRecipeLink{Frozen Cilantro Rice}, \mainRecipeLink{Tomato Soup}, or \mainRecipeLink{Charro Beans}
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item \textbf{Labels:} Write the part, count, recipe, and date on every bag.
      \item \textbf{Thawing:} Move a bag to the fridge the night before. Raw chicken keeps 9 months frozen.
      \item \textbf{Skin and Livers:} Throw away the extra skin and the livers. Skin makes the stock greasy, and livers make it bitter.
\end{itemize}

\makeRecipeTitle{Whole Chicken Plan}
\section*{The Plan}

\begin{enumerate}

      \item \textbf{Break Down the Birds} \\
            Break both birds into breasts, thighs, drumsticks, wings, and carcasses. Bone the thighs, and put the bones, necks, and wing tips with the carcasses.

      \item \textbf{Bag and Freeze} \\
            Slice the 2 Tuscan breasts in half horizontally into thin cutlets so they cook through in that recipe's time. Pat the thighs and breasts dry, press the air out of each bag, label it, and freeze it flat. Leave the wings and drums out for tonight.

      \item \textbf{Start the Stock} \\
            Get the carcasses simmering for \recipeLink{Frozen Chicken Stock}{Frozen Chicken Stock} (3--4 hours). It mostly cooks itself while you do everything else. Short on time? Freeze the carcasses and make the stock another day.

      \item \textbf{Dinner} \\
            While the stock simmers, make the \recipeLink{Lemon Pepper Wings and Drums}{Lemon Pepper Wings \& Drums}. Strain and chill the stock before bed.

      \item \textbf{Next Day} \\
            Skim the fat off the stock and portion it out. Keep 8 cups in the fridge and make the \recipeLink{Chicken Noodle Soup}{Chicken Noodle Soup} with the 2 whole breasts.

      \item \textbf{Through the Week} \\
            Pull the thighs for \mainRecipeLink{Chipotle Meal Prep} and the cutlets for \mainRecipeLink{Creamy Tuscan Chicken} as you need them. Both recipes are in the main recipe collection.

\end{enumerate}
