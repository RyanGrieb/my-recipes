\makeRecipeTitle{Charro Beans}
\recipeServings{8--10 (as a side)}

\section*{Ingredients}

\textbf{For the Beans}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Dried Pinto Beans – 1 lb (454g), rinsed and sorted
      \item[\cut] White Onion – ½ medium, peeled and halved
      \item[\cut] Garlic Cloves – 6, peeled
      \item[\measure] Water – 10 cups
      \item[\measure] Kosher Salt – 2 tsp
\end{itemize}

\textbf{Charro Base \& Meats}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\cut] Bacon – 8 oz, diced
      \item[\cut] Cooked Ham – 1 cup, diced (optional)
      \item Mexican Chorizo – 4 oz, casings removed (optional)
      \item[\cut] White Onion – 1 large, diced
      \item[\cut] Garlic Cloves – 3, minced
      \item[\cut] Jalapeño Peppers – 2 large, seeded and diced
      \item[\cut] Roma Tomatoes – 3, diced
      \item[\cut] Chipotle Pepper in Adobo – 1, minced (optional)
      \item[\measure] Chicken Broth – ½ cup (for deglazing)
\end{itemize}

\textbf{Seasonings \& Finishing}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\measure] Cumin – 2 tsp
      \item[\measure] Mexican Oregano – ½ tsp
      \item[\measure] Black Pepper – ½ tsp
      \item[\cut] Fresh Cilantro – ½ cup, chopped
\end{itemize}

\makeRecipeTitle{Charro Beans}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything (20 minutes)} \\
            Rinse and sort the beans, halve the half onion, and peel 6 garlic cloves for the pot. Dice the bacon, ham, large onion, and tomatoes, mince the other 3 garlic cloves and the chipotle, seed and dice the jalapeños, and chop the cilantro. Measure out everything marked with a spoon.

      \item \textbf{Cook the Beans (1.5--2 hours)} \\
            Put the beans, onion halves, peeled garlic, water, and salt in a large Dutch oven and bring to a boil. Cover and simmer on medium-low until the beans are tender (1.5--2 hours), adding water as needed to keep them submerged. Discard the onion and garlic.

      \item \textbf{Create Bean Paste (5 minutes)} \\
            Off the heat, blend 1 cup of the beans and 1 cup of the hot broth until completely smooth. \textbf{Safety Note:} remove the lid's center cap, cover the hole with a folded towel, and start on low. Transfer the remaining beans and broth to a large heatproof bowl.

      \item \textbf{Render Bacon \& Cook Meats (12--15 minutes)} \\
            In the empty pot, cook the bacon over medium heat until the fat renders and the edges are crispy (8--10 minutes), then remove with a slotted spoon. \textbf{If using chorizo:} brown it in the bacon fat, breaking it apart (4--5 minutes), then remove. \textbf{If using ham:} warm it through (1--2 minutes), then remove. Pour off all but 3 tablespoons of fat.

      \item \textbf{Sauté Aromatics (12--13 minutes)} \\
            Sauté the onion in the bacon fat until softened (5 minutes). Add the garlic and jalapeños and cook until fragrant (1 minute). Add the tomatoes and chipotle (if using) and cook until the tomatoes soften and release their juices (5--7 minutes).

      \item \textbf{Deglaze \& Combine (22--25 minutes)} \\
            Pour in the chicken broth and scrape up the browned bits (1 minute). Return the meats, the reserved beans and broth, and the bean paste to the pot, rinsing the blender into it with ½ cup water. Stir in the cumin, oregano, and pepper. Bring to a gentle boil, then cover and simmer on low, stirring occasionally (15--20 minutes).

      \item \textbf{Finish \& Serve} \\
            Off the heat, stir in the cilantro. Season with salt, starting with 1 teaspoon since the meats are already salty. Serve hot with your choice of garnishes.

\end{enumerate}
