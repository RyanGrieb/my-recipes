\makeRecipeTitle{Butter Chicken}
\recipeServings{4--5}

\section*{Ingredients}

\subsection*{For the Chicken Marinade:}

\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\cut] Boneless, Skinless Chicken Thighs or Breasts – 28 oz, cubed
      \item[\measure] Plain Yogurt – ½ cup
      \item[\cut] Garlic Minced – 1½ tbsp
      \item[\cut] Ginger Minced – 1 tbsp
      \item[\measure] Garam Masala – 2 tsp
      \item[\measure] Turmeric – 1 tsp
      \item[\measure] Ground Cumin – 1 tsp
      \item[\measure] Red Chili Powder – 1 tsp (Kashmiri recommended)
      \item[\measure] Salt – 1 tsp
\end{itemize}

\subsection*{For the Sauce:}

\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\measure] Olive Oil – 2 tbsp
      \item Ghee or Butter – 2 tbsp ghee or 1 tbsp butter + 1 tbsp oil
      \item[\cut] Large Onion – 1 (sliced or chopped)
      \item[\cut] Green Bell Peppers – 3, diced
      \item[\cut] Garlic Minced – 1½ tbsp
      \item[\cut] Ginger Minced – 1 tbsp
      \item[\measure] Ground Cumin – 1½ tsp
      \item[\measure] Garam Masala – 1½ tsp
      \item[\measure] Ground Coriander – 1 tsp
      \item Crushed Tomatoes – 14 oz
      \item[\measure] Red Chili Powder – 1 tsp (adjust to taste, use Kashmiri)
      \item[\measure] Salt – 1¼ tsp (or to taste)
      \item[\measure] Thickened Cream – 1 cup (heavy cream or evaporated milk to save calories)
      \item[\cut] Fresh Cilantro – to garnish, chopped (optional)
      %\item Kasoori Methi – ½ tsp (dried fenugreek leaves)
\end{itemize}

\newpage
\makeRecipeTitle{Butter Chicken}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything} \\
            Cut the chicken into bite-sized pieces. Mince all the garlic and ginger, slice the onion, and dice the bell peppers, then measure out everything marked with a spoon.

      \item \textbf{Marinate Chicken} \\
            In a bowl, mix plain yogurt, garlic, ginger, garam masala, turmeric, cumin, chili powder, and salt. Add chicken pieces and mix so it is coated evenly. Allow to marinate for at least 30 minutes (or overnight in the fridge for deeper flavor).

      \item \textbf{Sear Chicken} \\
            Heat a large skillet or pan over medium-high heat and add oil. Cook the marinated chicken in batches, searing until browned on all sides. Set aside.

      \item \textbf{Sauté Onions and Aromatics} \\
            In the same skillet, melt butter or ghee and sauté onions until softened. Add garlic, ginger, and remaining spices (garam masala, ground coriander, and cumin) and cook until fragrant.

      \item \textbf{Build the Sauce} \\
            Stir in crushed tomatoes, chili powder, and salt. Let the tomatoes cook for about 10--15 minutes until they are nearly dry in your pan, stirring occasionally until the sauce thickens and becomes a deep brown red color.

      \item \textbf{Add Cream and Season} \\
            Add cream to the sauce in the skillet and stir well. Taste and adjust the salt.

      \item \textbf{Simmer Chicken and Peppers} \\
            Add the seared chicken with its juices and the diced bell peppers to the sauce. Simmer for 8--10 minutes, until the chicken is cooked through, the peppers are tender but still have some bite, and the sauce is thick and bubbling.

      \item \textbf{Serve} \\
            Garnish with chopped cilantro (if using) and serve with rice or garlic naan for a complete meal!

\end{enumerate}
