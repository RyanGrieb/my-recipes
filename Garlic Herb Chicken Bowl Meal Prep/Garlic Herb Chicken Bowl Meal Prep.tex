\section*{Ingredients}

\textbf{The Chicken}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Chicken Thighs – 2.2 lbs (1kg), boneless, skinless, and diced
      \item Onion Powder – 1 tbsp (8g)
      \item Garlic Powder – 2½ tsp (8g)
      \item Smoked Paprika – 1 tbsp (8g)
      \item Ground Cumin – 1 tbsp (8g)
      \item Dried Oregano – 1 tbsp (4g)
      \item Honey – 1 tbsp (20g)
      \item Tomato Paste – 3 tbsp (50g)
      \item Lemon – 1 (juiced)
      \item Extra Virgin Olive Oil – 1 tbsp (15ml)
      \item Plain Greek Yogurt – 1 tbsp (20g)
      \item Salt and Pepper – to taste
\end{itemize}

\textbf{Spiced Pilaf Rice}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Basmati Rice – 1½ cups (300g), rinsed
      \item Unsalted Butter – 1 tbsp (14g)
      \item Yellow Onion – 1, diced
      \item Ground Turmeric – 2½ tsp (8g)
      \item Smoked Paprika – 1½ tsp (4g)
      \item Ground Cumin – 2 tsp (4g)
      \item Chicken Stock – 2½ cups (600ml)
      \item Salt – to taste
\end{itemize}

\section*{Instructions}

\begin{enumerate}

      \item \textbf{Marinate the Chicken} \\
            Combine the chicken with the garlic powder, onion powder, oregano, paprika, cumin, honey, tomato paste, lemon juice, olive oil, yogurt, salt, and pepper. Mix well and set aside to marinate while you prep everything else.

      \item \textbf{Cook the Chicken} \\
            Heat a large pan or griddle over medium-high heat with a splash of oil. Cook the chicken in batches (don't overcrowd the pan) for 8--10 minutes, until charred and cooked through to 165$^\circ$F (74$^\circ$C). Set aside to rest.

      \item \textbf{Make the Pilaf Rice} \\
            While the chicken cooks, melt the butter in a pot over low heat. Add the onion and cook until soft (4--5 minutes). Stir in the turmeric, paprika, cumin, and salt, and toast for 1 minute. Add the rinsed rice and stir until every grain is coated in the spiced butter. \\
            Pour in the chicken stock and bring to a boil. Cover, reduce to low, and cook for 12 minutes. Take the pot off the heat and let it rest with the lid on for 4 minutes.

      \item \textbf{Make the Tabbouleh} \\
            While the rice cooks, combine all the tabbouleh ingredients in a bowl. Refrigerate until needed.

      \item \textbf{Make the Garlic Sauce} \\
            Whisk together the yogurt, sriracha, garlic, paprika, onion powder, salt, and pepper until smooth. Taste and add more sriracha if you want extra heat.

      \item \textbf{Portion and Store} \\
            Divide the rice and chicken between 5 meal-prep containers (750ml each). Pack the tabbouleh and garlic sauce in separate small containers so they stay fresh.

\end{enumerate}
