\makeRecipeTitle{Vegetable Soup}
\recipeServings{4--5}

\section*{Ingredients}

\begin{itemize}
      \item Unsalted Butter – 1/4 cup
      \item[\cut] Yellow Onion, diced – 1 medium
      \item[\measure] All-Purpose Flour – 1/4 cup (30 g)
      \item[\measure] Broth (vegetable or chicken) – 16 ounces (454 g)
      \item[\measure] Whole Milk – 16 ounces (454 g)
      \item[\cut] Potatoes, diced into bite-size cubes – 1 pound (454 g)
      \item[\measure] Salt – 1/2 tsp
      \item[\measure] Black Pepper – 1/4 tsp
      \item[\measure] Italian Seasoning – 1 tsp
      \item Frozen Mixed Vegetables – 15 ounces (425 g)
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item \textbf{Potato Options:} Cube potatoes into small, bite-size pieces, about 1/2 inch or smaller. Yukon Gold, russet, or red potatoes all work well.
      \item \textbf{Storage:} Leftovers keep for 4--5 days in a tightly sealed container in the refrigerator. Reheat on the stovetop or in the microwave.
      \item \textbf{Veggie Substitutes:} Frozen mixed vegetables can be replaced with drained canned vegetables.
\end{itemize}

\makeRecipeTitle{Vegetable Soup}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything} \\
            Dice the onion and cut the potatoes into small, bite-size cubes. Measure out everything marked with a spoon.

      \item \textbf{Sauté Onion and Make Roux} \\
            In a stock pot over medium heat, melt the butter. Add the diced onion and cook for 3--4 minutes, until beginning to turn translucent. Sprinkle the flour over the onions and stir to coat; cook for 1 minute.

      \item \textbf{Add Liquids} \\
            Pour in the broth and milk, whisking continuously to remove any lumps.

      \item \textbf{Cook the Potatoes} \\
            Add the diced potatoes, salt, pepper, and Italian seasoning. Stir and cook for 10 minutes, stirring occasionally.

      \item \textbf{Add Frozen Vegetables} \\
            Stir in the frozen mixed vegetables. Continue cooking for 4--5 minutes more, or until the potatoes are tender and fall apart when pierced with a fork.

      \item \textbf{Serve} \\
            Remove from heat and serve warm.

\end{enumerate}
