\makeRecipeTitle{Chorizo \& Egg Muffins}
\recipeServings{5 (meal-prep containers, 12 muffins)}

\section*{Ingredients}

\textbf{The Filling}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\cut] Spanish Chorizo – 6.3 oz (180g), diced
      \item[\cut] Cherry Tomatoes – 5.3 oz (150g), halved
      \item[\measure] Olive Oil – 1 tsp (5ml), for the tomatoes
      \item Feta – 2.8 oz (80g), crumbled (Persian, Greek, or Danish)
      \item[\cut] Cheddar Cheese – 3.5 oz (100g), grated
\end{itemize}

\textbf{The Egg Base}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Eggs – 12 large
      \item[\measure] Milk – 1/3 cup (80ml)
      \item[\measure] Smoked Paprika – 1/2 tsp (2g)
      \item[\measure] Garlic Powder – 1/2 tsp (2g)
      \item Salt and Pepper – to taste
\end{itemize}

\textbf{To Serve}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\cut] Flat-Leaf Parsley – a small handful, chopped
      \item Feta – extra, crumbled, to taste
      \item Hot Sauce – to taste
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item \textbf{Chorizo:} Use cured Spanish chorizo (firm, like salami), not raw Mexican chorizo.
      \item \textbf{Storage:} Keeps up to 5 days in the refrigerator, or 2 months in the freezer. Thaw frozen muffins in the fridge overnight before reheating.
\end{itemize}

\makeRecipeTitle{Chorizo \& Egg Muffins}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything} \\
            Dice the chorizo, halve the cherry tomatoes, grate the cheddar, and chop the
            parsley. Measure out everything marked with a spoon.

      \item \textbf{Preheat the Oven} \\
            Set the oven to 350$^\circ$F and let it heat while you cook the filling.

      \item \textbf{Render the Chorizo} \\
            Add the chorizo to a pan over medium heat and cook until the fat has fully rendered
            and the edges are crispy (3--4 minutes). Lift the chorizo out with a slotted spoon
            and set it aside, leaving all the fat in the pan.

      \item \textbf{Blister the Tomatoes} \\
            In a separate pan, heat the olive oil over medium-high heat. Add the tomatoes
            cut-side down and cook, without stirring, until blistered underneath (2--3 minutes).
            Toss once, cook 1 more minute, then set aside.

      \item \textbf{Grease the Tin} \\
            Brush every hole of a 12-hole muffin tin generously with the chorizo fat, coating the
            base, sides, and rim. If the fat runs out, finish with a little olive oil.

      \item \textbf{Fill the Tin} \\
            Whisk the eggs, milk, smoked paprika, garlic powder, salt, and pepper until combined
            and slightly frothy. Divide the chorizo evenly between the 12 holes, add the
            tomatoes, crumble over the feta, and scatter over half the cheddar. Pour in the egg
            mix to fill each hole only 3/4 full, since the eggs rise as they bake. Top with the
            rest of the cheddar.

      \item \textbf{Bake} \\
            Bake until the eggs are set, the tops are golden, and the cheese is bubbling
            (18--22 minutes). A slight wobble in the center is fine; they finish setting as they
            rest. Let them rest in the tin (5 minutes), then run a knife around each one and lift
            it out.

      \item \textbf{Portion and Serve} \\
            Divide the 12 muffins between 5 containers (3 in two of them, 2 in the other three)
            and top with the parsley and extra feta. Keep the hot sauce separate until serving.
            Reheat in the microwave (45--60 seconds), or eat them cold.

\end{enumerate}
