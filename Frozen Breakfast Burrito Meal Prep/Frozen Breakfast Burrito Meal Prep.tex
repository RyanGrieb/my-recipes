\makeRecipeTitle{Frozen Breakfast Burrito Meal Prep}

\section*{Ingredients}

\textbf{The Filling}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Ground Turkey – 1 lb (93/7 lean)
      \item Avocado Oil – 1 tbsp
      \item Whipped Low-Fat Cottage Cheese – 1 lb (453g)
      \item Kale – ½ bunch (about 2 cups), chopped
      \item Large Eggs – 18, whisked
      \item Large Tortillas – 6 (the creator's Ultimate Tortillas, or store-bought)
\end{itemize}

\textbf{Spice Blend}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Chili Powder – 1 tbsp (8g)
      \item Smoked Paprika – ½ tsp (2g)
      \item Onion Powder – 1 tsp (3g)
      \item Garlic Powder – 1 tsp (3.5g)
      \item Ground Cumin – ½ tbsp (4g)
      \item Dried Oregano – ¼ tsp (a pinch)
      \item Black Pepper – 1 tsp (2g)
      \item Salt – 1 tsp (9g), or to taste
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item \textbf{Storage:} Wrap each burrito individually in parchment paper before storing it in the freezer.
      \item \textbf{Reheating (Best):} Move a burrito to the fridge the night before so it thaws. Microwave for 2 minutes, then let it rest 1--2 minutes before eating.
      \item \textbf{Reheating (From Frozen):} Microwave for 2 minutes, cut in half, then microwave for another 2 minutes.
\end{itemize}

\section*{Instructions}

\begin{enumerate}

      \item \textbf{Brown the Turkey} \\
            Heat the oil in a pan over medium heat. Add the ground turkey and cook, breaking it apart, until browned and cooked through. Stir in all of the spices and cook until fragrant (30--60 seconds).

      \item \textbf{Add Cottage Cheese and Kale} \\
            Stir in the cottage cheese and mix well. Add the kale (or other vegetables) and cook it down until the mixture starts to thicken.

      \item \textbf{Cool the Filling} \\
            Transfer the turkey mixture to a bowl or plate to cool, where it will continue to thicken.

      \item \textbf{Scramble the Eggs} \\
            Lightly scramble the eggs over medium-high heat until they are just cooked and creamy. Transfer to a container to cool.

      \item \textbf{Portion and Freeze} \\
            Divide the turkey filling and eggs between the 6 tortillas and wrap them tightly. Arrange the burritos in a baking dish or on a tray and freeze overnight. \\
            Once frozen solid, move them to freezer-safe containers or zip-top bags. They keep in the freezer for up to 3 months.

\end{enumerate}
