\makeRecipeTitle{Chorizo \& Chickpea Breakfast Frittata}
\recipeServings{8}

\section*{Ingredients}

\begin{itemize}
      \item[\cut] Chorizo Sausage – 7 oz (200g), diced
      \item Eggs – 5, beaten
      \item[\measure] Olive Oil – 1 tbsp (15ml)
      \item[\cut] Yellow Onion – 1, diced
      \item[\cut] Garlic Cloves – 3, sliced
      \item[\measure] Smoked Paprika – 1 tsp (2g)
      \item[\measure] Ground Cumin – 1 tsp (2g)
      \item[\cut] Red Bell Pepper – 1, diced
      \item Canned Chickpeas – 7 oz (200g), drained and rinsed
      \item Baby Spinach – 4 oz (120g), washed
      \item[\measure] Heavy Cream – 2 tbsp (30ml) (optional)
      \item[\cut] Flat-Leaf Parsley – 2 tbsp (5g), chopped, plus more to garnish
      \item Salt and Pepper – to taste
      \item Extra Virgin Olive Oil – to finish
      \item Snow Pea Sprouts – to garnish (optional)
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item \textbf{Pan:} You need a large ovenproof skillet (cast iron works well), since the frittata finishes under the broiler.
      \item \textbf{Storage:} Keeps for up to 4 days in an airtight container in the refrigerator. Reheat gently in a 300$^\circ$F oven for about 10 minutes, or eat it cold. Avoid the microwave; it makes the eggs rubbery.
\end{itemize}

\makeRecipeTitle{Chorizo \& Chickpea Breakfast Frittata}
\section*{Instructions}

\begin{enumerate}

      \item \textbf{Prep Everything} \\
            Dice the chorizo, onion, and bell pepper, slice the garlic, chop all the parsley, and beat the eggs. Measure out everything marked with a spoon.

      \item \textbf{Preheat the Broiler} \\
            Set the oven to broil (375$^\circ$F) and let it heat while you cook the filling.

      \item \textbf{Sauté Aromatics} \\
            Place a large ovenproof pan over medium-high heat. Add the olive oil, onion, and garlic and sauté until just soft and fragrant (3--4 minutes).

      \item \textbf{Render the Chorizo} \\
            Add the chorizo, smoked paprika, and cumin, and season with salt and pepper. Sauté until the chorizo releases its paprika-tinted oil (a full 2 minutes).

      \item \textbf{Add the Vegetables} \\
            Add the bell pepper and chickpeas and sauté until soft (5 minutes). Pile the spinach on top and let it soften (about 45 seconds), then stir it in and let it finish wilting.

      \item \textbf{Add the Eggs} \\
            Pour in the beaten eggs, cream (if using), and chopped parsley. Fold gently so everything is coated in egg, then cook until the eggs start to firm up around the edges (about 2 minutes). \\
            \textbf{Chef's Tip:} Don't overmix once the eggs go in; gentle folding keeps the frittata light rather than dense and rubbery.

      \item \textbf{Broil} \\
            Move the pan under the broiler and cook until the frittata is set through, golden, and puffed (2--3 minutes).

      \item \textbf{Rest and Turn Out} \\
            Let the frittata rest for the full 5 minutes so it firms up and slices cleanly. Run a spatula around the edges to loosen it, then carefully tip it out onto a serving plate or cutting board.

      \item \textbf{Serve} \\
            Slice into portions and top with parsley, snow pea sprouts, cracked black pepper, and a drizzle of extra virgin olive oil. Serve hot, warm, or cold. Enjoy!

\end{enumerate}
