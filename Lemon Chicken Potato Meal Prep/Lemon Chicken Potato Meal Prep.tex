\makeRecipeTitle{Lemon Chicken \& Potato Meal Prep}
\recipeServings{5 (meal-prep containers)}

\section*{Ingredients}

\textbf{The Chicken}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item Chicken Thighs – 2.2 lbs (1kg), boneless and skinless
      \item[\measure] Smoked Paprika – 1 tbsp (8g)
      \item[\measure] Dried Oregano – ¾ tsp (1g)
      \item[\measure] Garlic Powder – 2 tsp (6g)
      \item[\cut] Garlic Cloves – 2, grated
      \item[\cut] Lemon – ½, zested and juiced
      \item[\measure] Olive Oil – 1 tbsp (15ml)
      \item Salt and Pepper – to taste
\end{itemize}

\textbf{Crispy Potatoes}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\cut] Baby Potatoes – 2.1 lbs (950g), diced
      \item[\measure] Smoked Paprika – 1¼ tbsp (10g)
      \item[\measure] Garlic Powder – 1¼ tsp (4g)
      \item[\measure] Olive Oil – 1 tbsp (15ml)
      \item Salt and Pepper – to taste
\end{itemize}

\textbf{To Serve}
\begin{itemize}[itemsep=0.1em, label={\raisebox{-0.1ex}{\large\textbullet}}]
      \item[\cut] Flat-Leaf Parsley – to garnish, chopped
      \item Sriracha – to serve
\end{itemize}

\section*{Notes}

\begin{itemize}
      \item \textbf{Storage:} Keeps for up to 4 days in the refrigerator. Microwave the chicken until hot, and reheat the potatoes on a sheet pan in a 400$^\circ$F oven for 8--10 minutes to bring back the crispiness.
\end{itemize}

\section*{Instructions}

\begin{enumerate}

      \item \textbf{Preheat} \\
            Set one oven rack in the upper third and one in the lower third, and preheat the oven to 410$^\circ$F (convection).

      \item \textbf{Marinate the Chicken} \\
            Season the chicken with the salt, pepper, smoked paprika, oregano, and garlic powder. Add the grated garlic, lemon zest, lemon juice, and olive oil, and mix until evenly coated. Let it marinate for as long as you can, even just while the oven preheats.

      \item \textbf{Season the Potatoes} \\
            In a bowl, season the potatoes with the salt, pepper, smoked paprika, and garlic powder. Add the olive oil and toss to coat.

      \item \textbf{Roast the Potatoes} \\
            Spread the potatoes onto a large lined sheet pan (18 $\times$ 13 in), making sure they're well spaced. Roast on the upper rack on their own to start crisping (20 minutes).

      \item \textbf{Add the Chicken} \\
            After the potatoes have roasted for 20 minutes, place the chicken on a second lined tray and slide it onto the lower rack, leaving the potatoes on the upper rack. Cook, turning the potatoes once after about 10 minutes, until the potatoes are golden and crisp and the chicken is cooked through to 165$^\circ$F (a further 20--25 minutes).

      \item \textbf{Rest and Slice} \\
            Let the chicken rest for 5 minutes. Slice into bite-sized pieces and toss through the tray juices to add back moisture and flavor.

      \item \textbf{Portion and Store} \\
            Divide the potatoes between 5 meal-prep containers, top with the sliced chicken, and garnish with parsley. Serve with sriracha on the side or drizzled over just before eating.

\end{enumerate}
